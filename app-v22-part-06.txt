function animateFromRects(first){Array.from(document.querySelectorAll("#rankList .player")).forEach(el=>{const f=first.get(el);if(!f)return;const l=el.getBoundingClientRect();const dx=f.left-l.left,dy=f.top-l.top;if(Math.abs(dx)>0.5||Math.abs(dy)>0.5){el.animate([{transform:`translate(${dx}px,${dy}px)`},{transform:"translate(0,0)"}],{duration:190,easing:"cubic-bezier(.2,.8,.2,1)"})}})}
function moveDraggedRelative(target,e){
 const dragged=dragState.el;if(!dragged||target===dragged)return;
 if(rankPos!=="ALL"){const targetPlayer=players[Number(target.dataset.index)];const draggedPlayer=players[dragState.index];if(!targetPlayer||!draggedPlayer||targetPlayer.position!==draggedPlayer.position)return;if(target.parentElement!==dragged.parentElement) target.parentElement.appendChild(dragged)}
 const rect=target.getBoundingClientRect();const before=e.clientY < rect.top+rect.height/2;const first=captureRects();if(before)target.parentNode.insertBefore(dragged,target);else target.parentNode.insertBefore(dragged,target.nextSibling);animateFromRects(first)
}
function wireDrag(){document.querySelectorAll('#rankList .player[draggable=true]').forEach(r=>{
  r.addEventListener("pointerdown",()=>r.classList.add("holding"));r.addEventListener("pointerup",()=>r.classList.remove("holding"));r.addEventListener("pointerleave",()=>r.classList.remove("holding"));
  r.addEventListener("dragstart",e=>{dragState.el=r;dragState.index=Number(r.dataset.index);const p=players[dragState.index];dragState.startPos=rankPos==="ALL"?p.overall:p.posRank;dragState.startTier=p.tier;r.classList.remove("holding");r.classList.add("dragging");e.dataTransfer.effectAllowed="move";e.dataTransfer.setData("text/plain",r.dataset.index);transparentDragImage(e)});
  r.addEventListener("dragover",e=>{e.preventDefault();moveDraggedRelative(r,e)});r.addEventListener("drop",e=>{e.preventDefault();e.stopPropagation();commitVisualOrder()});r.addEventListener("dragend",()=>{r.classList.remove("holding");if(dragState.el)dragState.el.classList.remove("dragging");commitVisualOrder()})
 })}
