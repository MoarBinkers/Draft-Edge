async function refreshCurrentAdp(){
 document.getElementById("liveText").textContent="Updating current ADP…";
 try{
  let data=await sleeper("https://api.sleeper.app/v1/players/nfl");
  let arr=[];
  Object.entries(data).forEach(([id,p])=>{
   if(!p||typeof p.search_rank!=="number"||!(p.search_rank>0))return;
   let fps=p.fantasy_positions||[];let pos=fps.find(x=>MARKET_POS.includes(x))||p.position;
   if(!MARKET_POS.includes(pos))return;
   let full=(p.full_name||((p.first_name||"")+" "+(p.last_name||"")).trim());
   if(!full)return;
   arr.push({id,p,pos,full})
  });
  arr.sort((a,b)=>a.p.search_rank-b.p.search_rank);
  let posCounts={},changed=0,newPool=[];
  arr.forEach((x,i)=>{
   posCounts[x.pos]=(posCounts[x.pos]||0)+1;
   let hit={id:x.id,rank:i+1,posRank:posCounts[x.pos],team:x.p.team||"FA",pos:x.pos,searchRank:x.p.search_rank};
   let old=market[x.full]?.rank,move=old==null?0:old-hit.rank;
   if(old!=null&&old!==hit.rank){changed++;history[x.full]=history[x.full]||[];history[x.full].push({t:Date.now(),rank:hit.rank});if(history[x.full].length>250)history[x.full].shift()}
   market[x.full]={...hit,move};
   if(POS.includes(x.pos)&&newPool.length<450)newPool.push({name:x.full,position:x.pos,team:x.p.team||"FA",bye:"—",tier:null,tags:[],note:"",drafted:false})
  });
  sleeperPool=newPool;save();let now=new Date();
  document.getElementById("liveText").textContent="Current ADP updated "+now.toLocaleTimeString([], {hour:"numeric",minute:"2-digit"});
  document.getElementById("adpStatus").textContent=changed?changed+" player"+(changed===1?"":"s")+" moved since your last refresh":"No rank changes since your last refresh";
  renderEverything()
 }catch(e){document.getElementById("liveText").textContent="Couldn’t update ADP";document.getElementById("adpStatus").textContent=e.message}
}
